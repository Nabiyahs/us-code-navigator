        function openBook(code, year, title, org, color) {
            const popup = document.getElementById('book-popup');
            const badge = document.getElementById('code-badge');
            const codeTitle = document.getElementById('code-title');
            const codeOrg = document.getElementById('code-org');
            
            badge.textContent = `${code} - ${year}`;
            codeTitle.textContent = title;
            codeOrg.textContent = org;
            
            popup.classList.add('active');
            document.body.style.overflow = 'hidden';
        }
        
        function closeBook() {
            const popup = document.getElementById('book-popup');
            popup.classList.remove('active');
            document.body.style.overflow = 'auto';
        }
        
        document.getElementById('book-popup').addEventListener('click', function(e) {
            if (e.target === this) {
                closeBook();
            }
        });
        
        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape') {
                closeBook();
            }
        });
